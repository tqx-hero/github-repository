#### 漏桶的实现

基于`alarm(3)+pause(3)` 实现，用来控制流量，控制单位秒时间内可以处理多少字节。

下方的demo是使用定时器实现在终端输出一个文件，每秒十个字节。

```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <time.h>
#include <errno.h>
#include <signal.h>
#define CPS 10
#define BUF_SIZE CPS
//实现慢放的功能,使用信号的方式代替sleep()

static volatile int loop = 1;
static void alrm_handler(int signum){
    loop = 0;
    alarm(1);
}

//漏桶的实现
int main(int argc,char **argv){
    int sfd = -1,dfd = 1;
    char buf[BUF_SIZE];
    ssize_t bytes,res;
    if(argc < 2){
        fprintf(stderr,"Usage param <argv>\n");
        exit(EXIT_FAILURE);
    }

    do{
        sfd = open(argv[1],O_RDONLY);
        if(sfd < 0){
            if(errno != EINTR){
                perror("open argv[1] error");
                goto fail_ret;
            }
        }
    }while(sfd < 0);

    //注册信号
    if(signal(SIGALRM,alrm_handler) == SIG_ERR){
        perror("signal()");
        goto fail_ret;
    }
    alarm(1);

    while(1){
        while(loop)
            pause();
        loop = 1;
        
        while((bytes = read(sfd,buf,sizeof(buf))) < 0)
        {
            if(errno == EINTR)
                continue;
            perror("read error");
            goto fail_ret;
        }
        if(!bytes)
            break;
        res = 0;
        while(1){
            res = write(dfd,buf + res,bytes);
            if(res <= 0){
                if(res < 0 && errno == EINTR)
                    continue;
                perror("write error");
                goto fail_ret;
            }
            if(res >= bytes)
                break;
            bytes -= res;
        }
    }

    close(sfd);
    return 0;

fail_ret:
    if(sfd >= 0)
        close(sfd);
    exit(EXIT_FAILURE);
}
```



#### 令牌桶：

###### 令牌桶，即当出现意外中断，在这里指定时器到时，没有数据可以处理，就攒下token，等到有数据到来，进行处理时就按照token的个数进行相应次数的处理，而不是像漏桶那样，每次固定处理多少数据，这样做可以增加进程的处理效率。

```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <time.h>
#include <errno.h>
#include <signal.h>
//设置token的最大上限
#define TOKEN_MAX 50
#define CPS 10
#define BUF_SIZE CPS

//令牌桶功能
//令牌数
static volatile int token = 0;

//SIGALRM信号捕获
static void alrm_handler(int signum){
    if(++token > TOKEN_MAX)//计数器到时，token自增，且不超过最大数量
        token = TOKEN_MAX;
    alarm(1);
}

int main(int argc,char **argv){
    int sfd = -1,dfd = 1;
    char buf[BUF_SIZE];
    ssize_t bytes,res;
    if(argc < 2){
        fprintf(stderr,"Usage param <argv>\n");
        exit(EXIT_FAILURE);
    }

    do{
        sfd = open(argv[1],O_RDONLY);
        if(sfd < 0){
            if(errno != EINTR){
                perror("open argv[1] error");
                goto fail_ret;
            }
        }
    }while(sfd < 0);

    //注册信号
    if(signal(SIGALRM,alrm_handler) == SIG_ERR){
        perror("signal()");
        goto fail_ret;
    }
    alarm(1);

    while(1){

        //如果没有token，阻塞在这里
        while(!token)
            pause();
        --token;

        while((bytes = read(sfd,buf,sizeof(buf))) < 0)
        {
            if(errno == EINTR){
                continue;
            }
            perror("read error");
            goto fail_ret;
        }

        if(!bytes)
            break;
        res = 0;

        while(1){
            res = write(dfd,buf + res,bytes);
            if(res <= 0){
                if(res < 0 && errno == EINTR)
                    continue;
                perror("write error");
                goto fail_ret;
            }
            if(res >= bytes)
                break;
            bytes -= res;
        }
    }

    close(sfd);
    return 0;

fail_ret:
    if(sfd >= 0)
        close(sfd);
    exit(EXIT_FAILURE);
}
```
